// D6 named checker example. Registration is not execution readiness.
// App parameters supply "path"; the service mapping selects the web Task.
package checker

import (
	"context"
	lifecycle "github.com/jay94ks/seafield/lifecycle/v1"
)

func Check(ctx context.Context, in lifecycle.Request, probe lifecycle.Probe) (lifecycle.Result, error) {
	r, err := probe.File(ctx, lifecycle.FileProbe{
		Service: "web", Path: in.Parameters["path"], Kind: "regular",
	})
	if err != nil {
		return lifecycle.Result{Schema: 1, Status: lifecycle.Unknown, Reason: "file-query-error"}, err
	}
	if !r.Exists || !r.Match {
		return lifecycle.Result{Schema: 1, Status: lifecycle.Fail, Reason: "marker-not-ready"}, nil
	}
	return lifecycle.Result{Schema: 1, Status: lifecycle.Pass, Reason: "marker-ready"}, nil
}
